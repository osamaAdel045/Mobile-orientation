### Auth Flow Implementation

```tsx
// core/storage/auth.ts — Token management with MMKV + Keychain
import { MMKV } from 'react-native-mmkv';
import Keychain from 'react-native-keychain';

const storage = new MMKV({ id: 'auth-storage', encryptionKey: 'com.mkh.app' });

export const authStorage = {
  async setTokens(accessToken: string, refreshToken: string) {
    storage.set('accessToken', accessToken);
    await Keychain.setGenericPassword('refresh', refreshToken, {
      service: 'com.mkh.refresh',
      accessible: Keychain.ACCESSIBLE.WHEN_UNLOCKED_THIS_DEVICE_ONLY,
    });
  },
  getAccessToken: () => storage.getString('accessToken'),
  async getRefreshToken(): Promise<string | null> {
    const creds = await Keychain.getGenericPassword({ service: 'com.mkh.refresh' });
    return creds?.password ?? null;
  },
  async clear() {
    storage.delete('accessToken');
    storage.clearAll();
    await Keychain.resetGenericPassword({ service: 'com.mkh.refresh' });
  },
};

// features/auth/store.ts — Zustand auth store
interface AuthState {
  user: User | null;
  isAuthenticated: boolean;
  isLoading: boolean;
  login: (email: string, password: string) => Promise<void>;
  logout: () => Promise<void>;
  restoreSession: () => Promise<void>;
}

export const useAuthStore = create<AuthState>()((set, get) => ({
  user: null,
  isAuthenticated: false,
  isLoading: true,

  restoreSession: async () => {
    try {
      const token = authStorage.getAccessToken();
      if (token) {
        const user = await apiClient.getCurrentUser(token);
        set({ user, isAuthenticated: true, isLoading: false });
      } else {
        set({ isLoading: false });
      }
    } catch {
      await authStorage.clear();
      set({ user: null, isAuthenticated: false, isLoading: false });
    }
  },

  login: async (email, password) => {
    const { tokens, user } = await apiClient.login(email, password);
    await authStorage.setTokens(tokens.access, tokens.refresh);
    set({ user, isAuthenticated: true });
  },

  logout: async () => {
    await authStorage.clear();
    set({ user: null, isAuthenticated: false });
  },
}));
```

### Data Flow: Feed with Offline Support

```tsx
// features/feed/hooks/useFeed.ts
export function useFeed() {
  const queryClient = useQueryClient();

  // React Query manages server state with offline persistence
  const feedQuery = useInfiniteQuery({
    queryKey: ['feed'],
    queryFn: ({ pageParam = 1 }) => apiClient.getFeed({ page: pageParam, limit: 20 }),
    getNextPageParam: (lastPage) => lastPage.nextCursor,
    staleTime: 5 * 60 * 1000,         // Consider data fresh for 5 minutes
    gcTime: 24 * 60 * 60 * 1000,      // Keep in cache for 24 hours
  });

  // WatermelonDB sync on connectivity change
  const { isConnected } = useNetInfo();
  const database = useDatabase();

  useEffect(() => {
    if (isConnected && feedQuery.data) {
      synchronize({
        database,
        pullChanges: async ({ lastPulledAt }) => {
          const response = await apiClient.getFeedSync({ lastPulledAt });
          return { changes: response.changes, timestamp: response.timestamp };
        },
        pushChanges: async ({ changes }) => {
          await apiClient.pushFeedChanges(changes);
        },
      });
    }
  }, [isConnected, feedQuery.dataLastUpdated]);

  return {
    posts: feedQuery.data?.pages.flatMap((p) => p.items) ?? [],
    fetchNextPage: feedQuery.fetchNextPage,
    hasNextPage: feedQuery.hasNextPage,
    isFetchingNextPage: feedQuery.isFetchingNextPage,
    isLoading: feedQuery.isLoading,
    refetch: feedQuery.refetch,
    isOffline: !isConnected,
  };
}
```

### Push Notification Setup

```tsx
// features/notifications/setup.ts
import notifee, { AndroidImportance, EventType } from '@notifee/react-native';
import messaging from '@react-native-firebase/messaging';
import { navigationRef } from '@/core/navigation';

export async function setupNotifications() {
  const authStatus = await messaging().requestPermission();
  if (authStatus !== messaging.AuthorizationStatus.AUTHORIZED) return;

  const fcmToken = await messaging().getToken();
  await apiClient.registerPushToken(fcmToken);

  // Foreground notification display
  notifee.onForegroundEvent(({ type, detail }) => {
    if (type === EventType.PRESS && detail.notification?.data) {
      handleNotificationNavigation(detail.notification.data);
    }
  });

  // Background tap handler (app opened from notification)
  messaging().onNotificationOpenedApp((remoteMessage) => {
    handleNotificationNavigation(remoteMessage.data);
  });
}

function handleNotificationNavigation(data: Record<string, any>) {
  if (data?.type === 'new_content') {
    navigationRef.navigate('Content', { contentId: data.contentId });
  } else if (data?.type === 'comment_reply') {
    navigationRef.navigate('Post', { postId: data.postId, focusReply: true });
  }
}
```

<Callout title="This reference app proves RN's production viability — with caveats">
The Mobile Knowledge Hub app demonstrates that RN handles complex production requirements: offline sync, push notifications, typed navigation, CI/CD, native Turbo Modules, and multi-layered testing. The pain points that emerged: WatermelonDB sync conflict resolution required significant native expertise, pushing new animated transitions required Reanimated rather than core Animated, and the CI build pipeline (especially iOS archive) is 3x slower than a pure native project. These are manageable trade-offs, not blockers.
</Callout>

### What This Reference App Reveals

**RN Strengths in production:** Rapid feature iteration (a new screen in hours, not days), single-team productivity (one engineer owns a feature across both platforms), OTA critical hotfixes deployed without app review, TypeScript ecosystem integration (Zod for runtime validation, React Query for caching), and a hiring pool that made building a 4-person team feasible in weeks.

**RN Weaknesses in production:** WatermelonDB sync conflict resolution required native Kotlin/Swift code, pushing custom animated transitions went beyond core Animated (required Reanimated), CI build time for iOS archiving is 3x longer than a native Swift project, debugging native crashes requires Xcode/Android Studio proficiency that JS developers lack, and the 12-hour Sentry source map upload + Hermes symbolication setup was unexpectedly painful.
