    ### MVVM in Production

    MVVM dominates Android because `ViewModel` survives configuration changes and exposes `StateFlow` for lifecycle-safe observation.

    ```kotlin
    data class OrderUiState(
        val isLoading: Boolean = true,
        val orders: List<Order> = emptyList(),
        val error: String? = null
    )

    class OrderViewModel(
        private val orderRepository: OrderRepository,
        private val savedStateHandle: SavedStateHandle
    ) : ViewModel() {
        private val _uiState = MutableStateFlow(OrderUiState())
        val uiState: StateFlow<OrderUiState> = _uiState.asStateFlow()

        init { loadOrders() }

        private fun loadOrders() {
            viewModelScope.launch {
                _uiState.update { it.copy(isLoading = true) }
                orderRepository.getOrders()
                    .onSuccess { orders -> _uiState.update { it.copy(isLoading = false, orders = orders) } }
                    .onFailure { e -> _uiState.update { it.copy(isLoading = false, error = e.message) } }
            }
        }
        fun retry() = loadOrders()
    }
    ```

    <Callout title="StateFlow over LiveData">
      StateFlow is the modern replacement for LiveData. It requires an initial value, is cold-start safe,
      and integrates with Kotlin coroutines and Flow operators. LiveData lacks structured concurrency
      support — always prefer StateFlow in new code.
    </Callout>

    ### MVI with Sealed Classes

    Model-View-Intent models every user action as an `Intent` processed through a reducer, making state changes predictable and testable.

    ```kotlin
    sealed interface OrderIntent {
        data object LoadOrders : OrderIntent
        data class SearchOrders(val query: String) : OrderIntent
        data class ConfirmDelete(val orderId: String) : OrderIntent
    }

    data class OrderListState(
        val orders: List<Order> = emptyList(),
        val isLoading: Boolean = false
    )

    class MviOrderViewModel(private val repository: OrderRepository) : ViewModel() {
        private val _state = MutableStateFlow(OrderListState())
        val state: StateFlow<OrderListState> = _state.asStateFlow()
        private val _effects = Channel<OrderSideEffect>(Channel.BUFFERED)
        val effects: Flow<OrderSideEffect> = _effects.receiveAsFlow()

        fun process(intent: OrderIntent) = when (intent) {
            OrderIntent.LoadOrders -> loadOrders()
            is OrderIntent.SearchOrders -> searchOrders(intent.query)
            is OrderIntent.ConfirmDelete -> confirmDelete(intent.orderId)
        }

        private fun loadOrders() {
            viewModelScope.launch {
                _state.update { it.copy(isLoading = true) }
                repository.getOrders().fold(
                    onSuccess = { orders -> _state.update { it.copy(isLoading = false, orders = orders) } },
                    onFailure = { _effects.send(OrderSideEffect.ShowSnackbar(it.message ?: "Error")) }
                )
            }
        }
    }

    sealed interface OrderSideEffect {
        data class ShowSnackbar(val message: String) : OrderSideEffect
        data class NavigateToDetail(val orderId: String) : OrderSideEffect
    }
    ```

    ### Clean Architecture Layers

    Clean Architecture separates code into **data**, **domain**, and **presentation** layers. The domain layer is pure Kotlin — no Android dependencies — making it fully unit-testable.

    ```kotlin
    // Domain layer (pure Kotlin)
    class GetActiveOrdersUseCase(private val orderRepository: OrderRepository) {
        suspend operator fun invoke(): Result<List<Order>> =
            orderRepository.getOrders().map { orders -> orders.filter { it.status == OrderStatus.ACTIVE } }
    }

    interface OrderRepository {
        suspend fun getOrders(): Result<List<Order>>
    }

    // Data layer
    class OrderRepositoryImpl(
        private val remoteSource: OrderRemoteDataSource,
        private val localSource: OrderLocalDataSource
    ) : OrderRepository {
        override suspend fun getOrders(): Result<List<Order>> = runCatching {
            val cached = localSource.getCachedOrders()
            if (cached.isNotEmpty()) return@runCatching cached
            val remote = remoteSource.fetchOrders()
            localSource.cacheOrders(remote)
            remote
        }
    }
    ```

    ### Common Pitfalls

    - **Leaking ViewModels into domain**: Use cases must never reference `ViewModel` or `Context`. Declare repository interfaces in the domain layer.
    - **Giant ViewModels**: One ViewModel per feature screen to avoid SRP violations.
    - **Shared mutable state**: Use `update {}` or `compareAndSet {}` on `MutableStateFlow` to prevent race conditions.
    - **Over-architecting**: Not every screen needs a use case. Direct repository-to-ViewModel wiring is fine for CRUD-only features.
