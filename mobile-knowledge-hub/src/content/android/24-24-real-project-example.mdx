    ### Reference Architecture: TaskFlow App

    TaskFlow is a task management app demonstrating production-ready Android architecture. It combines recommended Jetpack libraries with modern Kotlin patterns in a modular, testable codebase.

    ### Tech Stack

    | Layer | Technology | Purpose |
    |---|---|---|
    | UI | Jetpack Compose + Material 3 | Declarative UI with dynamic color |
    | Navigation | Navigation Compose (type-safe) | Type-safe routes via kotlinx.serialization |
    | Architecture | MVVM + Clean Architecture | Separated concerns, testable layers |
    | DI | Hilt | Constructor injection, ViewModel scoping |
    | Network | Retrofit + OkHttp + Kotlinx Serialization | Type-safe HTTP with auth interceptors |
    | Database | Room + DataStore (Proto) | SQLite for entities, proto for preferences |
    | Async | Kotlin Coroutines + Flow | Structured concurrency, reactive streams |
    | Auth | Credential Manager + Biometric | Passkey and biometric authentication |
    | Push | Firebase Cloud Messaging | Remote notifications with data payloads |
    | Background | WorkManager | Deferrable, guaranteed execution |
    | CI/CD | GitHub Actions | Build, lint, test, and deploy |
    | Quality | Detekt, KtLint, Baseline Profiles | Static analysis, startup optimization |

    ### ViewModel with SavedStateHandle and Flow

    ```kotlin
    {
    @HiltViewModel
    class TasksViewModel @Inject constructor(
        private val getTasksUseCase: GetTasksUseCase,
        private val toggleTaskUseCase: ToggleTaskUseCase,
        private val savedStateHandle: SavedStateHandle
    ) : ViewModel() {

        private val _events = Channel<TaskUiEvent>(Channel.BUFFERED)
        val events: Flow<TaskUiEvent> = _events.receiveAsFlow()

        val filter: StateFlow<TaskFilter> = savedStateHandle
            .getStateFlow("filter", TaskFilter.ALL)

        val uiState: StateFlow<TasksUiState> = combine(
            getTasksUseCase(), filter
        ) { tasks, activeFilter ->
            TasksUiState(tasks = tasks.filteredBy(activeFilter), filter = activeFilter)
        }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), TasksUiState())

        fun toggleCompletion(taskId: String) {
            viewModelScope.launch {
                toggleTaskUseCase(taskId)
                _events.send(TaskUiEvent.TaskUpdated)
            }
        }
    }
    }
    ```

    ### Retrofit Service with Kotlinx Serialization

    ```kotlin
    {
    @Module
    @InstallIn(SingletonComponent::class)
    object NetworkModule {
        @Provides @Singleton
        fun provideOkHttpClient(authInterceptor: AuthInterceptor): OkHttpClient =
            OkHttpClient.Builder()
                .addInterceptor(authInterceptor)
                .addInterceptor(HttpLoggingInterceptor().apply {
                    level = HttpLoggingInterceptor.Level.BODY
                })
                .connectTimeout(30, TimeUnit.SECONDS)
                .readTimeout(30, TimeUnit.SECONDS)
                .build()

        @Provides @Singleton
        fun provideRetrofit(client: OkHttpClient): Retrofit = Retrofit.Builder()
            .baseUrl(BuildConfig.API_BASE_URL)
            .client(client)
            .addConverterFactory(json.asConverterFactory("application/json".toMediaType()))
            .build()
    }
    }
    ```

    ### WorkManager for Background Sync

    ```kotlin
    {
    @HiltWorker
    class SyncWorker @AssistedInject constructor(
        @Assisted context: Context,
        @Assisted params: WorkerParameters,
        private val taskRepository: TaskRepository
    ) : CoroutineWorker(context, params) {

        override suspend fun doWork(): Result = withContext(Dispatchers.IO) {
            try {
                taskRepository.syncPendingChanges()
                Result.success()
            } catch (e: IOException) {
                if (runAttemptCount < 3) Result.retry() else Result.failure()
            }
        }

        companion object {
            fun enqueueSync(context: Context) {
                val constraints = Constraints.Builder()
                    .setRequiredNetworkType(NetworkType.CONNECTED).build()
                val request = OneTimeWorkRequestBuilder<SyncWorker>()
                    .setConstraints(constraints)
                    .setBackoffCriteria(BackoffPolicy.EXPONENTIAL, 30, TimeUnit.SECONDS)
                    .build()
                WorkManager.getInstance(context).enqueue(request)
            }
        }
    }
    }
    ```
